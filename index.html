async function loadMenu() {

    const response = await fetch(url);

    if (!response.ok) {
      throw new Error(`HTTP Error: ${response.status}`);

    console.error("Menu loading failed:", error.message);
  }
}

function mapDishDto(dto) {
  return {

    name: dto.name,

CRUD · CREATE

POST — ობიექტის გაგზავნა სერვერზე

async function addToCart(dishId, quantity = 1) {
  const response = await fetch("/api/v1/learning/cart", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "X-Session-Id": "student_token_123"
    },
    body: JSON.stringify({ dishId, quantity })
  });

  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  return response.json();
}

body

JavaScript Object-ს პირდაპირ არ ვაგზავნით. ჯერ JSON string უნდა გახდეს.

Content-Type

ვატყობინებთ სერვერს, რომ body-ის ფორმატი JSON არის.

POST / PUT — IMPORTANT

Object ≠ JSON String

JS Object

{ dishId: 1, quantity: 1 }

ჩვენს JavaScript-ში არსებული მონაცემი.

→

JSON String

'{"dishId":1,"quantity":1}'

HTTP body-ში გასაგზავნი ტექსტური ფორმა.

ამ გარდაქმნას აკეთებს JSON.stringify(). უკუღმა, Response-ის JSON-ის წაკითხვას ასრულებს response.json().
